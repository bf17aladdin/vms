Option 1: WebSocket (Real-time Browser Streaming)

   const socket = io('http://localhost:8000');
   
   socket.emit('start_stream', {
       camera_id: 'front_door',
       camera_name: 'Front Door'
   });
   
   socket.on('detection_result', (data) => {
       console.log('Motion:', data.motion.detected);
       console.log('Objects:', data.objects);
   });

Option 2: Python Async Processing
────────────────────────────────
   from vms.backend.services.async_frame_pipeline import get_pipeline
   
   pipeline = get_pipeline('cam_1', 'Front Door')
   results = await pipeline.process_frame(frame, db=session)
   
   print(f"Motion: {results['motion']['detected']}")
   print(f"Objects: {len(results['objects'])}")

Option 3: Multi-Camera Parallel
───────────────────────────────
   processor = get_async_processor()
   processor.add_camera('cam_1', 'Front Door')
   processor.add_camera('cam_2', 'Garage')
   
   results = await processor.process_frames_parallel({
       'cam_1': frame1,
       'cam_2': frame2
   })
